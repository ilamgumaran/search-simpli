"""S6-T23 verifier: an edge folder of invented files, run through every surface.

usage: python3 -I edge_probe.py <scratch> <head-src> <main-src> <head searchd> <main searchd> <head dylib> <main dylib>

Writes <scratch>/edge/ and prints, per file, what each surface did with it:
head and main `index --update` (JSON report and list-files), the head's
`search` stderr line, the head's ss_index_folder (ctypes, full and update),
and the Python reference's walk (`_iter_text_files`, `not_read_counts`) of
the head and of main. Content is invented.
"""

E = S / "edge"
if E.exists():
    raise SystemExit("edge folder exists; remove it by its literal path first")
E.mkdir()

W = b"heron "  # 6 bytes, a word to pad with
files: dict[str, bytes] = {}


def pad(n: int, unit: bytes = W) -> bytes:
    return (unit * (n // len(unit) + 1))[:n]

def run(cmd):
    return subprocess.run(cmd, capture_output=True)


def walked_regular():
    out = []
    for dirpath, dirnames, filenames in os.walk(os.fsencode(str(E))):
        rel_dir = os.path.relpath(dirpath, os.fsencode(str(E)))
        parts = [] if rel_dir == b"." else rel_dir.split(b"/")
        if any(p.startswith(b".") or p in (b".git", b".search", b".zig-cache", b"__pycache__", b"node_modules", b"zig-out") for p in parts):
            continue
        for f in filenames:
            full = os.path.join(dirpath, f)
            if os.path.islink(full) or not os.path.isfile(full) or f.startswith(b"."):
                continue
            out.append(b"/".join(parts + [f]))
    return sorted(out)


regular = walked_regular()
print(f"# regular files the walk keeps (my own walk, lstat, no dot, no ignored): {len(regular)}")

results = {}
for label, sd in (("head", head_sd), ("main", main_sd)):
    out = S / f"edge-idx-{label}"
    r = run([sd, "index", str(E), "--out", str(out), "--update"])
    results[label] = {"rc": r.returncode, "stdout": r.stdout.decode("utf-8", "replace"), "stderr": r.stderr.decode("utf-8", "replace")}
    lf = run([sd, "list-files", str(out), "--json", "--page-size", "1000"])
    results[label]["list"] = lf.stdout.decode("utf-8", "replace") if lf.returncode == 0 else f"rc={lf.returncode} {lf.stderr!r}"
    rr = run([sd, "index", str(E), "--out", str(S / f"edge-full-{label}")])
    results[label]["full_line"] = (rr.returncode, rr.stdout.decode("utf-8", "replace").strip(), rr.stderr.decode("utf-8", "replace").strip())
    sr = run([sd, "search", str(E), "kestrel", "--top-k", "3"])
    results[label]["search"] = (sr.returncode, sr.stderr.decode("utf-8", "replace").strip())

for label in ("head", "main"):
    r = results[label]
    print(f"\n## {label} searchd")
    print(f"index --update rc={r['rc']} stderr={r['stderr'].strip()!r}")
    print(f"index (full) {r['full_line']}")
    print(f"search stderr {r['search']}")
    if r["rc"] == 0:
        rep = json.loads(r["stdout"])
        print("report:", json.dumps({k: rep[k] for k in rep if k not in ("terms", "postings")}, ensure_ascii=True))

indexed = {}
for label in ("head", "main"):
    try:
        listing = json.loads(results[label]["list"])
        indexed[label] = sorted(f["path"] for f in listing.get("files", []))
    except Exception:
        indexed[label] = None
        print(f"{label} list-files: {results[label]['list'][:300]!r}")


def py_walk(src):
    code = (
        "import sys, json; sys.path.insert(0, sys.argv[1]);"
        "from pathlib import Path; from search_platform import core;"
        "root = Path(sys.argv[2]);"
        "paths = [p.relative_to(root).as_posix() for p in core._iter_text_files(root)];"
        "nr = core.not_read_counts(root) if hasattr(core, 'not_read_counts') else None;"
        "print(json.dumps({'paths': paths, 'not_read': nr}, ensure_ascii=True))"
    )
    r = subprocess.run([sys.executable, "-I", "-c", code, str(Path(src) / "src"), str(E)], capture_output=True)
    if r.returncode != 0:
        return {"error": r.stderr.decode("utf-8", "replace")[-600:]}
    return json.loads(r.stdout)


py = {"head": py_walk(head_src), "main": py_walk(main_src)}
for label in ("head", "main"):
    if "error" in py[label]:
        print(f"\n## Python reference ({label}) FAILED: {py[label]['error']}")
    else:
        print(f"\n## Python reference ({label}): {len(py[label]['paths'])} read, not_read={py[label]['not_read']}")

# ABI
abi = {}
for label, lib in (("head", head_lib), ("main", main_lib)):
    L = ctypes.CDLL(lib)
    L.ss_index_folder.restype = ctypes.c_void_p
    L.ss_index_folder.argtypes = [ctypes.c_char_p, ctypes.c_char_p, ctypes.c_char_p]
    L.ss_last_error.restype = ctypes.c_char_p
    L.ss_free.argtypes = [ctypes.c_void_p]
    rows = []
    for mode, opts in (("full", b"{}"), ("update", b'{"update": true}')):
        p = L.ss_index_folder(str(S / f"edge-abi-{label}").encode(), str(E).encode(), opts)
        if not p:
            rows.append((mode, None, L.ss_last_error().decode()))
            continue
        text = ctypes.string_at(p).decode()
        L.ss_free(p)
        rows.append((mode, json.loads(text), None))
    abi[label] = rows
    for mode, rep, err in rows:
        if rep is None:
            print(f"\n## ss_index_folder {label} {mode}: NULL {err}")
        else:
            print(f"\n## ss_index_folder {label} {mode}: " + json.dumps({k: rep[k] for k in rep if k not in ("terms", "postings")}, ensure_ascii=True))

# per-file table
