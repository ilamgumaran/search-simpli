## probe/gen2.zig
const std = @import("std");
fn Gen(comptime T: type) type {
    return struct {
        pub const t = T;
        test "generic counted" {
            try std.testing.expect(true);
        }
    };
}
test "instantiates twice" {
    try std.testing.expect(Gen(u8).t == u8);
    try std.testing.expect(Gen(u16).t == u16);
}
## probe/holder.zig
const std = @import("std");
const Unused = struct {
    test "holder nested fails" {
        try std.testing.expectEqual(@as(u32, 1), @as(u32, 2));
    }
};
const Holder = struct {
    const alias = Unused;
    pub const n: u32 = 3;
};
test "names Holder" {
    _ = Holder;
    try std.testing.expectEqual(@as(u32, 3), Holder.n);
}

## tiny build: a step with zero tests, then a warm rerun (zig build test --summary all; the summary is on stderr, stdout empty)
const std = @import("std");
pub fn build(b: *std.Build) void {
    const target = b.standardTargetOptions(.{});
    const optimize = b.standardOptimizeOption(.{});
    const test_step = b.step("test", "t");
    const z = b.addTest(.{ .name = "zero", .root_module = b.createModule(.{ .root_source_file = b.path("src/zero.zig"), .target = target, .optimize = optimize }) });
    test_step.dependOn(&b.addRunArtifact(z).step);
    const o = b.addTest(.{ .name = "one", .root_module = b.createModule(.{ .root_source_file = b.path("src/one.zig"), .target = target, .optimize = optimize }) });
    test_step.dependOn(&b.addRunArtifact(o).step);
}
