"""Verifier probe: if each returned passage were widened to its whole file, how would passage@k and complete move?
The scorer credits any passage that holds the range, whatever its size (passage bytes are reported apart)."""
import json, re, subprocess, sys
from pathlib import Path
B, RUN, J = sys.argv[1], Path(sys.argv[2]), json.load(open(sys.argv[3]))
sys.path.insert(0, sys.argv[4]); import text_search_score as sc
nlines = {str(p.relative_to(RUN/"corpus")): len(re.split(r"\r?\n", p.read_bytes().decode().rstrip("\n"))) for p in (RUN/"corpus").rglob("*") if p.is_file()}
for field in ("both", "content"):
    real = wide = 0; rc = wc = 0; rb = wb = 0
    for q in J["questions"]:
        if q["kind"] == "no-answer": continue
        out = subprocess.run([B, "evidence", str(RUN/"index"), sc.query_of(q), "--search-field", field, "--top-k", "10"], capture_output=True, check=True).stdout
        ps = sc.passages_of(json.loads(out))
        seen, wps = set(), []
        for p in ps:
            if p["path"] in seen: continue
            seen.add(p["path"]); wps.append({"path": p["path"], "start_line": 1, "end_line": nlines[p["path"]], "content": (RUN/"corpus"/p["path"]).read_text()})
        a, b = sc.score_question(q, ps, 0), sc.score_question(q, wps, 0)
        real += a["passage_rank"] is not None and a["passage_rank"] <= 10; wide += b["passage_rank"] is not None and b["passage_rank"] <= 10
        rc += a["complete"]; wc += b["complete"]; rb += a["passage_bytes"]; wb += b["passage_bytes"]
    print(f"[{field}] passage@10 {real} -> {wide}; complete {rc} -> {wc}; mean passage bytes {rb/117:.0f} -> {wb/117:.0f}")
