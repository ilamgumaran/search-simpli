"""Verifier probe: print the passages searchd returns for chosen questions (or raw queries)."""
import json, subprocess, sys
from pathlib import Path
B, RUN, J = sys.argv[1], Path(sys.argv[2]), json.load(open(sys.argv[3]))
qs = {q["id"]: q for q in J["questions"]}
sys.path.insert(0, sys.argv[4])
import text_search_score as sc
for arg in sys.argv[5:]:
    field = "both"
    if arg.startswith("content:"): field, arg = "content", arg[8:]
    if arg.startswith("raw="):
        q = {"id": "raw", "kind": "no-answer", "question": arg[4:], "files": [], "parts": []}
    else:
        q = qs[arg]
    query = sc.query_of(q)
    out = subprocess.run([B, "evidence", str(RUN/"index"), query, "--search-field", field, "--top-k", "10"], capture_output=True, check=True).stdout
    ps = sc.passages_of(json.loads(out))
    row = sc.score_question(q, ps, len(out))
    print(f"== {q['id']} [{field}] query={query!r} rank file={row.get('file_rank')} passage={row.get('passage_rank')} parts={row.get('parts_found')}/{row.get('parts')} returned={len(ps)} bytes={len(out)}")
    for r in [r for p in q["parts"] for r in p["ranges"]]:
        print(f"   judged {r['path']}:{r['start_line']}-{r['end_line']}")
    for i, p in enumerate(ps, 1):
        first = p["content"].strip().splitlines()[0][:70] if p["content"].strip() else ""
        print(f"   {i:2d}. {p['path']}:{p['start_line']}-{p['end_line']}  | {first}")
