"""Verifier probe: for every question, compare the scorer's 'holds whole' rule with a 'partial overlap' rule,
and record passage sizes (lines). Uses the index surface."""
import json, subprocess, sys, statistics
from pathlib import Path
B, RUN, J = sys.argv[1], Path(sys.argv[2]), json.load(open(sys.argv[3]))
sys.path.insert(0, sys.argv[4])
import text_search_score as sc
def overlaps(p, r):
    return p["path"] == r["path"] and p["start_line"] <= r["end_line"] and r["start_line"] <= p["end_line"]
for field in ("both", "content"):
    diffs, sizes, raw = [], [], {}
    for q in J["questions"]:
        out = subprocess.run([B, "evidence", str(RUN/"index"), sc.query_of(q), "--search-field", field, "--top-k", "10"], capture_output=True, check=True).stdout
        ps = sc.passages_of(json.loads(out))
        sizes += [p["end_line"] - p["start_line"] + 1 for p in ps]
        if q["kind"] == "no-answer":
            continue
        whole = sum(1 for part in q["parts"] if any(sc.holds(p, r) for p in ps for r in part["ranges"]))
        part_ = sum(1 for part in q["parts"] if any(overlaps(p, r) for p in ps for r in part["ranges"]))
        if whole != part_:
            diffs.append((q["id"], whole, part_, len(q["parts"])))
    print(f"[{field}] passages: {len(sizes)}, lines per passage: median {statistics.median(sizes)}, max {max(sizes)}, "
          f">=20 lines: {sum(1 for s in sizes if s >= 20)}")
    print(f"[{field}] questions where partial overlap would find more parts than 'holds whole': {diffs}")
