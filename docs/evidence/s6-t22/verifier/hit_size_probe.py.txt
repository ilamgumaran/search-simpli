"""Verifier probe: the size in lines of the passage that scores each passage hit (first holding passage)."""
import json, subprocess, sys
from pathlib import Path
B, RUN, J = sys.argv[1], Path(sys.argv[2]), json.load(open(sys.argv[3]))
sys.path.insert(0, sys.argv[4])
import text_search_score as sc
for field in ("both", "content"):
    big = []
    for q in J["questions"]:
        if q["kind"] == "no-answer": continue
        out = subprocess.run([B, "evidence", str(RUN/"index"), sc.query_of(q), "--search-field", field, "--top-k", "10"], capture_output=True, check=True).stdout
        ps = sc.passages_of(json.loads(out))
        rs = [r for part in q["parts"] for r in part["ranges"]]
        for p in ps:
            if any(sc.holds(p, r) for r in rs):
                n = p["end_line"] - p["start_line"] + 1
                if n >= 15: big.append(f"{q['id']} {p['path']}:{p['start_line']}-{p['end_line']} ({n} lines)")
                break
    print(f"[{field}] first holding passage of 15+ lines: {len(big)}"); [print("   ", b) for b in big]
